async function loadTeamData() { try { let { data: profile } = await supabaseClient .from('profiles') .select('referral_code, team_income') .eq('id', currentUser.id) .maybeSingle(); let refCode = profile ? profile.referral_code : null; if (!refCode) { refCode = currentUser.id.substring(0, 8); await supabaseClient.from('profiles').update({ referral_code: refCode }).eq('id', currentUser.id); } // FIX: Replaced signup.html with login.html const currentFolder = window.location.href.substring(0, window.location.href.lastIndexOf('/')); generatedLink = `${currentFolder}/login.html?ref=${refCode}`; document.getElementById('referralLinkInput').value = generatedLink; // Set WhatsApp Direct Share Link const waText = encodeURIComponent(`Join Earnflow Seeds and get daily returns on seed investments! Register using my referral link: ${generatedLink}`); document.getElementById('whatsappShareBtn').href = `https://wa.me/?text=${waText}`; document.getElementById('teamIncome').innerText = '₹' + parseFloat((profile && profile.team_income) || 0).toFixed(2); // Fetch team members list const { data: members } = await supabaseClient .from('profiles') .select('email, created_at') .eq('referred_by', refCode) .order('created_at', { ascending: false }); const container = document.getElementById('teamMemberList'); document.getElementById('totalTeamCount').innerText = members ? members.length : 0; if (!members || members.length === 0) { container.innerHTML = `
No members joined yet. Share your referral link to earn commissions!
`; return; } container.innerHTML = members.map(m => { const rawPhone = (m.email || '').replace('@earnflow.app', ''); const maskedPhone = rawPhone.length === 10 ? rawPhone.substring(0, 4) + '****' + rawPhone.substring(8) : rawPhone; const joinDate = new Date(m.created_at).toLocaleDateString('en-IN', { day: '2-digit', month: 'short' }); return `

${maskedPhone}

Joined: ${joinDate}
Active
`; }).join(''); } catch (err) { // Fallback link on error const fallbackCode = currentUser.id.substring(0, 8); const currentFolder = window.location.href.substring(0, window.location.href.lastIndexOf('/')); generatedLink = `${currentFolder}/login.html?ref=${fallbackCode}`; document.getElementById('referralLinkInput').value = generatedLink; } }
